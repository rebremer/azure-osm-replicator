# ─────────────────────────────────────────────────────────────────────
# World run — full planet from planet.openstreetmap.org
#
# The URLs below match the script defaults; they're written out
# explicitly so switching *from* a Germany run only requires editing
# this file, not remembering the defaults.
# ─────────────────────────────────────────────────────────────────────

# 1 Log into the VM

ssh -i ~/osm-vm-key <<your user>>@<<your IP>>

# 2 Preflight — osm-update-on-boot.service must be enabled BEFORE running
#   init or update, otherwise the next VM boot won't pick up replication.
systemctl is-enabled osm-update-on-boot.service     # expect: enabled

# 3 Strip any prior region block (Germany OR world) from osm-env.sh, so
#   the file stays clean when you switch back and forth.
sudo sed -i \
  -e '/# Geofabrik regional load.*Germany/,/PBF_BLOB_PATH=initial\/germany-latest\.osm\.pbf/d' \
  -e '/# planet\.openstreetmap\.org.*full planet/,/PBF_BLOB_PATH=initial\/planet-latest\.osm\.pbf/d' \
  /etc/profile.d/osm-env.sh

cat /etc/profile.d/osm-env.sh

# 4 Persist world overrides — inherited by future systemd boot runs too.
sudo tee -a /etc/profile.d/osm-env.sh >/dev/null <<'EOF'

# planet.openstreetmap.org — full planet
export SOURCE_PBF_URL=https://planet.openstreetmap.org/pbf/planet-latest.osm.pbf
export REPLICATION_SERVER=https://planet.openstreetmap.org/replication/day/
export PBF_BLOB_PATH=initial/planet-latest.osm.pbf
EOF
. /etc/profile.d/osm-env.sh

cat /etc/profile.d/osm-env.sh

# 5 Kick off init with RESET_ALL=1 (drops tables, wipes nodes.bin,
#   clears replication blobs). RESET_ALL_PBF=1 is redundant on a
#   freshly deployed / emptied container, but doesn't hurt.
tmux new -s osm-init '
  export RESET_ALL=1
  export RESET_ALL_PBF=1
  { ./init-osm.sh
    echo "=== EXIT $? at $(date -u +%FT%TZ) ==="
  } 2>&1 | tee -a ~/init-osm-$(date -u +%Y%m%dT%H%M%SZ).log
'

tmux attach -t osm-init

# 6 After init: boot VM and confirm osm-update-on-boot.service actually fired on this boot

journalctl -u osm-update-on-boot.service -b --no-pager | tail -20

# 7 Check if state in database corresponds to planet state.txt

. /etc/profile.d/osm-env.sh
export PGPASSWORD=$(curl -fsS -H 'Metadata: true' \
  "http://169.254.169.254/metadata/identity/oauth2/token?api-version=2018-02-01&resource=https://vault.azure.net&client_id=${UAI_CLIENT_ID}" \
  | jq -r .access_token \
  | xargs -I{} curl -fsS -H "Authorization: Bearer {}" \
      "https://${KEY_VAULT_NAME}.vault.azure.net/secrets/${PG_PASSWORD_SECRET_NAME:-pg-admin-password}?api-version=7.4" \
  | jq -r .value)
psql -h "$PGHOST" -U "$PGUSER" -d "$PGDATABASE" -c \
  "SELECT property, value FROM osm2pgsql_properties WHERE property LIKE 'replication_%';"

# Compare with planet's current head:
curl -s https://planet.openstreetmap.org/replication/day/state.txt

# 8 (Continuously-running VMs) Install / refresh a daily timer that reuses
#   osm-update-on-boot.service. Change TIMER_TIME below and re-paste this
#   whole block — idempotent for both create and update. Systemd
#   guarantees the unit is exclusive, so an accidental boot at the same
#   time does not double-run.
TIMER_TIME="10:45"

sudo tee /etc/systemd/system/osm-update-daily.timer >/dev/null <<EOF
[Unit]
Description=Daily OSM replication update at ${TIMER_TIME} Europe/Berlin

[Timer]
# CET/CEST-aware — tzdata handles the twice-yearly DST shift automatically.
OnCalendar=*-*-* ${TIMER_TIME}:00 Europe/Berlin
# Fires at next boot if the VM was offline at ${TIMER_TIME} (e.g. stop/start
# cycle spanning that time).
Persistent=true
Unit=osm-update-on-boot.service

[Install]
WantedBy=timers.target
EOF

sudo systemctl daemon-reload
sudo systemctl enable osm-update-daily.timer    # no-op if already enabled
sudo systemctl restart osm-update-daily.timer   # picks up the new OnCalendar

# Verify
systemctl list-timers osm-update-daily.timer --all
systemctl is-enabled osm-update-daily.timer   # expect: enabled
systemctl is-active  osm-update-daily.timer   # expect: active

journalctl -u osm-update-daily.timer --since today --no-pager

# 9 See if update-osm.sh is running right now, and follow its output

# Running? 'activating' = in progress, 'inactive' = finished (or not yet fired)
systemctl is-active osm-update-on-boot.service

# Live tail of the current / most recent run
journalctl -u osm-update-on-boot.service -f

# Last-run summary + exit code (Ctrl-C out of the tail first)
systemctl status osm-update-on-boot.service --no-pager
